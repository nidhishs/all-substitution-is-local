\section{Preliminaries}
\label{sec:prelim}

We work within finite-action Bayesian decision problems \citep{DeGroot1970,Howard1966}.

\paragraph{Decision problem.}
An instance with features $x$ arrives from an input space $\mathcal{X}$.
The unknown state is $y \in \mathcal{Y} = \{1, \ldots, K\}$, $K \geq 2$.
The decision-maker chooses an action $a$ from a finite set $\mathcal{A}$, $|\mathcal{A}| \geq 2$, and receives reward $r_a(y)$ specified by a reward matrix $R \in \R^{|\mathcal{A}| \times K}$.

\paragraph{Belief and terminal value.}
A \emph{belief} $b \in \Delta^{K-1}$ is a probability vector over states.
A calibrated model produces $b_x = P(Y \mid x) \in \Delta^{K-1}$.
The \emph{terminal value} at belief $b$ is
\begin{equation}\label{eq:terminal-value}
  V(b) \;=\; \max_{a \in \mathcal{A}}\; r_a \cdot b,
  \qquad r_a \cdot b \;:=\; \textstyle\sum_y r_a(y)\, b(y).
\end{equation}
$V$ is piecewise-linear and convex on $\Delta^{K-1}$ as the pointwise maximum of affine functions (formalized as \texttt{terminalValue} in \texttt{Regret.lean}).
The \emph{model-optimal action} is $a_x = \argmax_{a \in \mathcal{A}} r_a \cdot b_x$.

\paragraph{Decision regions and facets.}
For each action $a$, the \emph{decision region} is
\[
  \mathcal{R}_a \;=\; \bigl\{ b \in \Delta^{K-1} : r_a \cdot b \geq r_{a'} \cdot b \;\text{ for all } a' \in \mathcal{A} \bigr\}.
\]
Each $\mathcal{R}_a$ is a convex polytope; the regions tile $\Delta^{K-1}$.
The relative boundary $\mathcal{R}_a \cap \mathcal{R}_{a'}$ is a \emph{decision facet}: the locus of beliefs at which the optimal action changes.

\paragraph{Human signal and updated belief.}
A recorded human signal $h$ (a radiology read, triage code, confidence rating, or expert prediction) is associated with instance $x$ in the dataset.
After observing $h$, the belief updates to $b_{x,h} = P(Y \mid x, h)$ via Bayes' rule (formalized as \texttt{posterior} in \texttt{Basics.lean}).
Because $h$ is recorded rather than freshly elicited, review costs can in principle be absorbed into the reward matrix as a constant offset; in our experiments (Section~\ref{sec:experiments}) we set review cost to zero to isolate the informational value of human review from its operational cost.

\paragraph{Value of information.}
The value of observing $h$ before acting, relative to acting on $b_x$ alone, is
\begin{equation}\label{eq:voi-def}
  \VoI(H \mid b_x) \;=\; \E_{h \mid x}\bigl[V(b_{x,h})\bigr] - V(b_x).
\end{equation}
By Jensen's inequality and the convexity of $V$, $\VoI \geq 0$.

\paragraph{Regret.}
The regret of action $a$ at belief $b$ is
\begin{equation}\label{eq:regret-def}
  \operatorname{regret}(a, b) \;=\; V(b) - r_a \cdot b \;\geq\; 0,
\end{equation}
with equality if and only if $a$ is optimal at $b$ (formalized as \texttt{regret\_eq\_zero\_iff} in \texttt{Regret.lean}).

\begin{remark}[Bayesian martingale]\label{rem:martingale}
The posterior is an unbiased estimate of the prior: $\E_{h \mid x}[b_{x,h}] = b_x$.
This is the law of total probability applied coordinate-wise (formalized as \texttt{prior\_eq\_weighted\_posterior} in \texttt{Basics.lean}).
As a consequence, for any fixed action $a$:
\begin{equation}\label{eq:martingale}
  \E_{h \mid x}[r_a \cdot b_{x,h}] \;=\; r_a \cdot b_x.
\end{equation}
This identity is the sole algebraic ingredient behind Theorem~\ref{thm:voi-br}.
\end{remark}
